\item Given equation is $\sin\theta = -1$

  $\Rightarrow \sin\theta = \sin\left(-\frac{\pi}{2}\right)$

  $\theta = \frac{n\pi}{4}, (2n + 1)\frac{\pi}{10}$ where $n\in I.$

  1)\frac{\pi}{2}$

  $x = \frac{2n\pi - \frac{\pi}{2}}{a - b}, \frac{(2n + 1)\pi + \frac{\pi}{2}}{a + b}$

\item $\cos y = -\cos x,~\sin y = 1 - \sin x.$ Squaring and adding, $1 = \cos^2x + 1 - 2\sin x + \sin^2x$

\item Squaring, $\sin^2x = 2\sin^2y,~3\cos^2x = 2\cos^2y.$ Adding, $\sin^2x + 3\cos^2x = 2\Rightarrow \cos^2x =

\item $2\sin x\sin4x = \cos3x - \cos5x$

\item We need $\tan x > 0,~\tan x\neq 1.$ Then $2 + 4\cos^2x = \tan^2x = \frac{1}{\cos^2x} - 1$

\item Let $s = \frac{x + y}{2},~d = \frac{x - y}{2}.$ $2\cos s\cos d - (2\cos^2s - 1) = \frac{3}{2}$

\item Let $p = \sin^2x, q = \cos^2x, p + q = 1.$ By $a^2 + b^2\geq\frac{(a + b)^2}{2},$

  $\therefore x = (2n + 1)\frac{\pi}{4},~y = 2m\pi + \frac{\pi}{2},$ where $m, n\in I.$

\item $x^2 + 2x\sin(xy) + 1 = (x + \sin(xy))^2 + \cos^2(xy) = 0$

\item If $\cos x > 0,$ then $2^{\cos x} > 1\geq|\sin x|:$ no solution. So $x\in\left[\frac{\pi}{2}, \frac{3\pi}{2}\right]$ or

\item Let $c = \cos2x.$ $\sin^6x + \cos^6x = 1 - \frac{3}{4}\sin^22x = \frac{1}{4} + \frac{3}{4}c^2$
