\subject{Challenging Problems}
\startitemize[n, 1*broad]
\item Since $\cos(180^\circ + x) = -\cos x,$ the terms $\cos k^\circ$ and $\cos(180 + k)^\circ$ cancel for $k = 1, 2, \ldots,
  179.$

  Only $\cos 180^\circ = -1$ remains. Thus, the sum $= -1.$

\item Since $\sin(90^\circ + x) = \cos x,$ we have $\sin^2k^\circ + \sin^2(90 + k)^\circ = 1.$

  $\sum_{k = 0}^{359}\sin^2k^\circ$ splits into $180$ such pairs ($k = 0, \ldots, 89$ and $k = 180, \ldots, 269$), so it equals
  $180.$

  Since $\sin 0^\circ = 0,$ the required sum $= 180.$

\item There are $90$ factors. For $91 \leq k \leq 179,$ $\tan k^\circ = -\tan(180 - k)^\circ,$ giving $45$ negative signs.

  So the product $= (-1)^{45}\left(\tan 1^\circ\tan 3^\circ \ldots \tan 89^\circ\right)^2.$

  Pairing $\tan k^\circ\tan(90 - k)^\circ = \tan k^\circ\cot k^\circ = 1$ and using $\tan 45^\circ = 1,$ the bracket is $1.$

  Answer: $-1.$

\item Let $t = \tan k^\circ;$ then $\tan(90 - k)^\circ = \frac{1}{t}.$

  $\frac{1}{1 + t} + \frac{1}{1 + \frac{1}{t}} = \frac{1}{1 + t} + \frac{t}{1 + t} = 1.$

  The $44$ pairs $(k, 90 - k),$ $k = 1, \ldots, 44,$ give $44$ and the middle term is $\frac{1}{1 + \tan 45^\circ} =
  \frac{1}{2}.$

  Answer: $\frac{89}{2}.$

\item For $0 \leq k \leq n - 1,$ $\cos\left(\theta + \frac{(k + n)\pi}{n}\right) = \cos\left(\pi + \theta +
  \frac{k\pi}{n}\right) = -\cos\left(\theta + \frac{k\pi}{n}\right).$

  Thus, the terms with indices $k$ and $k + n$ cancel, and the $2n$ terms cancel in $n$ pairs. Hence, the sum is $0.$

\item $\cos\frac{(n - k)\pi}{n} = \cos\left(\pi - \frac{k\pi}{n}\right) = -\cos\frac{k\pi}{n}.$

  Since $2m + 1$ is odd, $\cos^{2m + 1}\frac{(n - k)\pi}{n} = -\cos^{2m + 1}\frac{k\pi}{n}.$

  Pairing $k$ with $n - k$ cancels the terms; if $n$ is even, the unpaired middle term is $\cos^{2m + 1}\frac{\pi}{2} = 0.$
  Hence, the sum is $0.$

\item $\sin\frac{(2n - k)\pi}{2n} = \sin\left(\pi - \frac{k\pi}{2n}\right) = \sin\frac{k\pi}{2n},$ so the sum is $2\sum_{k =
    1}^{n - 1}\sin^2\frac{k\pi}{2n} + \sin^2\frac{\pi}{2}.$

  $\sin\frac{(n - k)\pi}{2n} = \sin\left(\frac{\pi}{2} - \frac{k\pi}{2n}\right) = \cos\frac{k\pi}{2n},$ so pairing $k$ with
  $n - k$ gives $\sum_{k = 1}^{n - 1}\sin^2\frac{k\pi}{2n} = \frac{n - 1}{2}$ (if $n$ is even, the middle term is
  $\sin^2\frac{\pi}{4} = \frac{1}{2}$).

  Hence, the sum $= 2.\frac{n - 1}{2} + 1 = n.$

\item $\tan\frac{(2n + 1 - k)\pi}{2n + 1} = \tan\left(\pi - \frac{k\pi}{2n + 1}\right) = -\tan\frac{k\pi}{2n + 1}.$

  The $2n$ terms cancel in the $n$ pairs $(k, 2n + 1 - k),$ $k = 1, \ldots, n.$ None of the angles equals $\frac{\pi}{2}$
  because $2n + 1$ is odd. Hence, the sum is $0.$

\item $\sin 2 = \sin(\pi - 2) = \sin 1.14,$ $\sin 3 = \sin(\pi - 3) = \sin 0.14,$ $\sin 4 = -\sin(4 - \pi) = -\sin 0.86,$ $\sin 5
  = -\sin(2\pi - 5) = -\sin 1.28.$
